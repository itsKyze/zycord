quickstart.ts — __INSTANCE_NAME__ SDK v9 / v10 Compatible 
// Connect a Bot to the Zycord WebSocket Gateway
const ws = new WebSocket("wss://" + location.host + "/?v=9&encoding=json");

ws.onopen = () => {
  ws.send(JSON.stringify({
    op: 2, // IDENTIFY
    d: {
      token: "Bot YOUR_ZYCORD_BOT_TOKEN",
      intents: 513, // GUILDS | GUILD_MESSAGES
      properties: { $os: "linux", $browser: "zycord-bot", $device: "zycord-bot" }
    }
  }));
};
// Send a Rich Embed Message via REST API v9
await fetch("/api/v9/channels/CHANNEL_ID/messages", {
  method: "POST",
  headers: {
    "Authorization": "Bot YOUR_ZYCORD_BOT_TOKEN",
    "Content-Type": "application/json"
  },
  body: JSON.stringify({
    content: "🚀 Deployment complete!",
    embeds: [{ title: "Build #402", color: 0x5865f2 }]
  })
});
// Exchange OAuth2 Authorization Code for Access Token
const res = await fetch("/api/v9/oauth2/token", {
  method: "POST",
  headers: { "Content-Type": "application/x-www-form-urlencoded" },
  body: new URLSearchParams({
    client_id: "YOUR_APP_ID",
    client_secret: "YOUR_APP_SECRET",

    code: authCode,
    redirect_uri: "https://your-app.example/callback"
